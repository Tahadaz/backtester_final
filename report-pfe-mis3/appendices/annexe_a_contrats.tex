\chapter{Contrats documentaires inter-stations}

\section{Contrat Data vers Signal}
\begin{verbatim}
Input attendu:
  - series OHLCV valides par symbole
  - index temporel strictement croissant
  - granularite 1D
Sortie:
  - dataset exploitable pour evaluation OOS
\end{verbatim}

\section{Contrat Signal vers Strategie}
\begin{verbatim}
Input attendu:
  - scores par famille/horizon
  - variantes representatives + robustesse
Sortie:
  - hypotheses de regles exploitables
\end{verbatim}

\section{Contrat Strategie vers Backtest}
\begin{verbatim}
Input attendu:
  - regles d'entree/sortie
  - parametres optimisables + bornes
  - contraintes de risque
Sortie:
  - run WFO completement parametre
\end{verbatim}

\section{Contrat Backtest vers Decision}
\begin{verbatim}
Input attendu:
  - metriques IS/OOS
  - WFE, statistiques, diagnostics
Sortie:
  - verdict de robustesse exploitable par le desk
\end{verbatim}
